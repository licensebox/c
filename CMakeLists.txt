cmake_minimum_required(VERSION 3.20)
project(licensebox VERSION 0.4.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(CURL 7.85 REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This initial C client supports Linux. Native adapters for other OSes are pending.")
endif()
add_library(licensebox src/licensebox.c src/json.c)
add_library(licensebox::c ALIAS licensebox)
set_target_properties(licensebox PROPERTIES EXPORT_NAME c VERSION ${PROJECT_VERSION} SOVERSION 0 POSITION_INDEPENDENT_CODE ON)
target_compile_features(licensebox PUBLIC c_std_11)
target_include_directories(licensebox PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(licensebox PRIVATE OpenSSL::Crypto CURL::libcurl PkgConfig::JSONC Threads::Threads m)
target_compile_options(licensebox PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  add_executable(licensebox-adapter tests/adapter.c)
  target_link_libraries(licensebox-adapter PRIVATE licensebox PkgConfig::JSONC)
  add_executable(licensebox-json-adapter tests/json_adapter.c)
  target_link_libraries(licensebox-json-adapter PRIVATE licensebox PkgConfig::JSONC)
  add_executable(licensebox-json-test tests/json_test.c)
  target_link_libraries(licensebox-json-test PRIVATE licensebox)
  add_test(NAME json-input-validation COMMAND licensebox-json-test)
endif()
install(TARGETS licensebox EXPORT licenseboxTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/licensebox.h include/licensebox_json.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT licenseboxTargets NAMESPACE licensebox:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/licensebox)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/licenseboxConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/licenseboxConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/licensebox)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/licenseboxConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/licenseboxConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/licenseboxConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/licensebox)
